10 REM probe-cfg: read-only, reads nextsync.cfg byte by byte (no end of line in the file)
20 CLS:PRINT "probe-cfg"
100 PRINT "1 DIM #5 TO n":ON ERROR GO TO 120
110 CLOSE #5:OPEN #5,"c:/sys/config/nextsync.cfg":DIM #5 TO n:PRINT "  size ";n:GO TO 130
120 ERROR TO e,l,s:PRINT "  err ";e;" stmt ";s
130 ON ERROR GO TO 140:CLOSE #5
140 ON ERROR :GO TO 200
200 PRINT "2 NEXT #5 TO b once":ON ERROR GO TO 220
210 CLOSE #5:OPEN #5,"c:/sys/config/nextsync.cfg":NEXT #5 TO b:PRINT "  byte ";b:GO TO 230
220 ERROR TO e,l,s:PRINT "  err ";e;" stmt ";s
230 ON ERROR GO TO 240:CLOSE #5
240 ON ERROR :GO TO 300
300 PRINT "3 NEXT #5 until error":LET a$="":LET c=0:ON ERROR GO TO 330
310 CLOSE #5:OPEN #5,"c:/sys/config/nextsync.cfg"
320 NEXT #5 TO b:LET c=c+1:IF c<40 AND b>0 THEN LET a$=a$+CHR$ b:GO TO 320
325 PRINT "  stopped, count ";c;" [";a$;"]":GO TO 340
330 ERROR TO e,l,s:PRINT "  err ";e;" stmt ";s;" count ";c;" [";a$;"]"
340 ON ERROR GO TO 350:CLOSE #5
350 ON ERROR :GO TO 400
400 PRINT "4 x=NEXT #5":ON ERROR GO TO 420
410 CLOSE #5:OPEN #5,"c:/sys/config/nextsync.cfg":LET x=NEXT #5:PRINT "  byte ";x:GO TO 430
420 ERROR TO e,l,s:PRINT "  err ";e;" stmt ";s
430 ON ERROR GO TO 440:CLOSE #5
440 ON ERROR :PRINT "press a key":PAUSE 0
