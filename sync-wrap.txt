50 OPEN #5,"syncwrap.cfg"
60 INPUT #5;LINE a$:INPUT #5;LINE b$:INPUT #5;LINE c$:INPUT #5;LINE d$

90 LET m=VAL a$:LET s=VAL b$:LET f=VAL c$:LET t=VAL d$:LET n=VAL e$:LET r=VAL f$:LET d=VAL g$

110 IF m<0 OR m>1 OR s<0 OR s>2 OR f<0 OR f>1 OR t<0 OR t>1 OR n<0 OR n>1 OR r<0 OR r>1 OR d<0 OR d>2 THEN GO TO 140
120 LET mo=m:LET sp=s:LET fc=f:LET tr=t:LET an=n:LET rt=r:LET dr=d:ON ERROR :GO TO 150
130 ON ERROR GO TO 140:CLOSE #5
140 ON ERROR :LET mo=0:LET sp=0:LET fc=1:LET tr=0:LET an=1:LET rt=1:LET dr=1
150 REM read the Server address saved by .sync5: no end of line in the file, so read DIM bytes
160 LET ad$="":ON ERROR GO TO 190
170 CLOSE #7:OPEN #7,"c:/sys/config/nextsync.cfg"
175 DIM #7 TO n:IF n>60 THEN LET n=60
176 IF n<1 THEN GO TO 180
177 FOR i=1 TO n:NEXT #7 TO b:IF b>32 THEN LET ad$=ad$+CHR$ b

180 CLOSE #7:ON ERROR :GO TO 200
190 ON ERROR GO TO 195:CLOSE #7
195 ON ERROR :GO TO 200
200 GO SUB 1000
205 REM key loop: the screen is redrawn line by line, never in full
210 PAUSE 0:LET k=CODE INKEY$:IF k>96 THEN LET k=k-32

230 IF k=83 THEN LET sp=sp+1:GO SUB 420:GO SUB 620:GO TO 310

270 IF k=82 THEN LET rt=1-rt:GO SUB 455:GO SUB 620:GO TO 310

285 IF k=65 THEN GO TO 800
286 IF k=13 THEN GO TO 700
290 IF k=88 THEN GO TO 990

340 CLOSE #4:ON ERROR :GO TO 210
350 ON ERROR GO TO 360:CLOSE #4
360 ON ERROR :GO TO 210

410 LET m$="Listen ":IF mo=1 THEN LET m$="Sync   "
415 PRINT AT 4,13;m$:RETURN

422 IF sp=2 THEN LET s$="fast   "
423 PRINT AT 5,13;s$:RETURN
430 LET f$="on ":IF fc=0 THEN LET f$="off"
431 PRINT AT 6,13;f$:RETURN
440 LET t$="on ":IF tr=0 THEN LET t$="off"
441 PRINT AT 7,13;t$:RETURN
450 LET n$="on ":IF an=0 THEN LET n$="off"
451 PRINT AT 8,13;n$:RETURN
455 LET r$="on ":IF rt=0 THEN LET r$="off"
456 PRINT AT 9,13;r$:RETURN
460 IF dr>2 THEN LET dr=0
461 LET d$="C: ":IF dr=1 THEN LET d$="D: "
462 IF dr=2 THEN LET d$="none"
463 PRINT AT 10,13;d$:RETURN
470 LET p$="not set":IF LEN ad$>0 THEN LET p$=ad$
471 LET p$=p$+"                           ":PRINT AT 3,4;p$( TO 27):RETURN
600 REM build the option string in c$
601 LET c$=""
602 IF mo=0 THEN LET o$="-L":GO SUB 650
603 IF mo=0 AND tr=0 THEN LET o$="-nv":GO SUB 650
604 IF sp=1 THEN LET o$="-s":GO SUB 650
605 IF sp=2 THEN LET o$="-f":GO SUB 650
606 IF fc=0 THEN LET o$="-nfc":GO SUB 650
607 IF an=0 THEN LET o$="-na":GO SUB 650
608 IF rt=0 THEN LET o$="-nr":GO SUB 650

620 REM show the command that ENTER will launch
621 GO SUB 600
622 LET p$="":IF mo=0 AND dr=0 THEN LET p$=".cd "+CHR$ 34+"C:"+CHR$ 34
623 IF mo=0 AND dr=1 THEN LET p$=".cd "+CHR$ 34+"D:"+CHR$ 34
624 LET p$=p$+"                                ":PRINT AT 16,0;p$( TO 32)
625 LET p$=".sync5 "+c$+"                                ":PRINT AT 17,0;p$( TO 32):RETURN

651 IF LEN c$>0 THEN LET c$=c$+" "
