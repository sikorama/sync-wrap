10 REM sync-wrap
20 LET mo=0:LET sp=0:LET fc=1:LET tr=0:LET an=1:LET rt=1:LET dr=1
30 REM load syncwrap.cfg: any problem keeps the defaults
40 ON ERROR GO TO 130
50 OPEN #5,"syncwrap.cfg"
60 INPUT #5;LINE a$:INPUT #5;LINE b$:INPUT #5;LINE c$:INPUT #5;LINE d$
70 INPUT #5;LINE e$:INPUT #5;LINE f$:INPUT #5;LINE g$
80 CLOSE #5
90 LET m=VAL a$:LET s=VAL b$:LET f=VAL c$:LET t=VAL d$:LET n=VAL e$:LET r=VAL f$:LET d=VAL g$
100 IF m<>INT m OR s<>INT s OR f<>INT f OR t<>INT t OR n<>INT n OR r<>INT r OR d<>INT d THEN GO TO 140
110 IF m<0 OR m>1 OR s<0 OR s>2 OR f<0 OR f>1 OR t<0 OR t>1 OR n<0 OR n>1 OR r<0 OR r>1 OR d<0 OR d>2 THEN GO TO 140
120 LET mo=m:LET sp=s:LET fc=f:LET tr=t:LET an=n:LET rt=r:LET dr=d:ON ERROR :GO TO 150
130 ON ERROR GO TO 140:CLOSE #5
140 ON ERROR :LET mo=0:LET sp=0:LET fc=1:LET tr=0:LET an=1:LET rt=1:LET dr=1
150 REM read the Server address saved by .sync5
160 LET ad$="":ON ERROR GO TO 190
170 OPEN #5,"c:/sys/config/nextsync.cfg"
180 INPUT #5;LINE ad$:CLOSE #5:ON ERROR :GO TO 200
190 ON ERROR GO TO 195:CLOSE #5
195 ON ERROR :GO TO 200
200 GO SUB 1000
210 PAUSE 0:LET k$=INKEY$:LET c=0
220 IF k$="m" OR k$="M" THEN LET c=1:LET mo=1-mo
230 IF k$="s" OR k$="S" THEN LET c=1:LET sp=sp+1:IF sp>2 THEN LET sp=0
240 IF k$="f" OR k$="F" THEN LET c=1:LET fc=1-fc
250 IF k$="t" OR k$="T" THEN LET c=1:LET tr=1-tr
260 IF k$="n" OR k$="N" THEN LET c=1:LET an=1-an
270 IF k$="r" OR k$="R" THEN LET c=1:LET rt=1-rt
280 IF k$="d" OR k$="D" THEN LET c=1:LET dr=dr+1:IF dr>2 THEN LET dr=0
285 IF k$="a" OR k$="A" THEN GO TO 500
286 IF k$=CHR$ 13 THEN GO TO 700
290 IF k$="x" OR k$="X" THEN GO TO 900
300 IF c=0 THEN GO TO 210
310 REM save syncwrap.cfg: a failed write is ignored
320 ON ERROR GO TO 350
330 OPEN #4,"o>syncwrap.cfg":PRINT #4;mo:PRINT #4;sp:PRINT #4;fc:PRINT #4;tr:PRINT #4;an:PRINT #4;rt:PRINT #4;dr
340 CLOSE #4:ON ERROR :GO TO 200
350 ON ERROR GO TO 360:CLOSE #4
360 ON ERROR :GO TO 200
500 CLS:PRINT "Server address"
510 PRINT "Current:"
520 IF LEN ad$=0 THEN PRINT "not set"
530 IF LEN ad$>0 THEN PRINT ad$
540 PRINT:PRINT "Not checked: type it right."
550 INPUT "New: ";LINE na$
560 IF LEN na$=0 THEN PRINT "Empty: refused":GO TO 620
570 IF na$(1)="-" THEN PRINT "No leading dash":GO TO 620
580 ON ERROR GO TO 610
590 LET c$=na$
600 .$ sync5 c$
605 ON ERROR :GO TO 150
610 ON ERROR :PRINT "Save failed"
620 PRINT "Press a key":PAUSE 0:GO TO 200
700 REM Launch: check the dot first
710 CLS:ON ERROR GO TO 790
720 OPEN #6,"c:/dot/sync5":CLOSE #6:ON ERROR
730 LET c$="":REM build the option string
731 IF mo=0 THEN LET o$="-L":GO SUB 1500
732 IF mo=0 AND tr=0 THEN LET o$="-nv":GO SUB 1500
733 IF sp=1 THEN LET o$="-s":GO SUB 1500
734 IF sp=2 THEN LET o$="-f":GO SUB 1500
735 IF fc=0 THEN LET o$="-nfc":GO SUB 1500
736 IF an=0 THEN LET o$="-na":GO SUB 1500
737 IF rt=0 THEN LET o$="-nr":GO SUB 1500
740 REM Drive, Listen mode only
741 IF mo=1 OR dr=2 THEN GO TO 770
742 IF dr=1 THEN GO TO 760
750 .cd "C:"
755 GO TO 770
760 .cd "D:"
770 POKE 23692,255
775 .$ sync5 c$
780 POKE 23692,255:GO TO 200
790 ON ERROR :PRINT "sync5 not found in c:/dot":PRINT "Press a key":PAUSE 0:GO TO 200
900 CLS:GO TO 9999
1000 REM draw the Summary screen
1010 IF mo=0 THEN LET m$="Listen"
1020 IF mo=1 THEN LET m$="Sync"
1030 IF sp=0 THEN LET s$="default"
1040 IF sp=1 THEN LET s$="slow"
1050 IF sp=2 THEN LET s$="fast"
1060 LET f$="on":IF fc=0 THEN LET f$="off"
1070 LET t$="on":IF tr=0 THEN LET t$="off"
1080 LET n$="on":IF an=0 THEN LET n$="off"
1090 LET r$="on":IF rt=0 THEN LET r$="off"
1100 IF dr=0 THEN LET d$="C:"
1110 IF dr=1 THEN LET d$="D:"
1120 IF dr=2 THEN LET d$="none"
1130 CLS
1140 PRINT "sync-wrap"
1150 PRINT
1160 PRINT "[A] Address"
1165 IF LEN ad$=0 THEN PRINT "    not set"
1166 IF LEN ad$>0 THEN PRINT "    ";ad$
1170 PRINT "[M] Mode:    ";m$
1180 PRINT "[S] Speed:   ";s$
1190 PRINT "[F] Flow:    ";f$
1200 PRINT "[T] Trace:   ";t$
1210 PRINT "[N] Anim:    ";n$
1220 PRINT "[R] Retro:   ";r$
1230 PRINT "[D] Drive:   ";d$
1240 PRINT
1250 PRINT "ENTER  Launch"
1260 PRINT "[X]    Exit"
1270 RETURN
1500 REM append option o$ to c$
1510 IF LEN c$>0 THEN LET c$=c$+" "
1520 LET c$=c$+o$:RETURN
9999 REM end
